     1→#!/bin/bash
     2→# Build the M2 delivery payload for the Next.js download page:
     3→#   - full git bundle at current tip (complete milestone history)
     4→#   - zero-dotfile buildable source tree zip + tar.gz twin
     5→#     (web scaffold + app/page.tsx + app/layout.tsx shims excluded)
     6→#   - copies of the M2 APK
     7→# Outputs land in public/ (served by the web app) with a backup in dist-master/.
     8→set -euo pipefail
     9→
    10→PROJECT=/home/z/my-project
    11→PUBLIC=$PROJECT/public
    12→DIST=$PROJECT/dist-master
    13→VERSION=v0.6.2-m2.6
    14→TOPDIR=PocketShell-$VERSION
    15→STAGE=$(mktemp -d)
    16→trap 'rm -rf "$STAGE"' EXIT
    17→
    18→cd "$PROJECT"
    19→TIP=$(git rev-parse --short HEAD)
    20→echo "== payload $VERSION @ git tip $TIP =="
    21→
    22→mkdir -p "$DIST" "$PUBLIC"
    23→
    24→# --- 1. git bundle: complete history, one file ---
    25→BUNDLE=$STAGE/pocketshell-m2.gitbundle
    26→git bundle create "$BUNDLE" --all
    27→git bundle verify "$BUNDLE" > /dev/null
    28→echo "bundle ok: $(du -h "$BUNDLE" | cut -f1)"
    29→
    30→# --- 2. RESTORE.txt (references the bundle for history) ---
    31→cat > "$STAGE/RESTORE.txt" << EOF
    32→PocketShell - source snapshot ($VERSION, git tip $TIP)
