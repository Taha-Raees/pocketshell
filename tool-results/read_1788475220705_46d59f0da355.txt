    70→/**
    71→ * Phase 3.3 "Home & System UI" — the PocketShell workspace
    72→ * (docs/PHASE-3.3-DESIGN.md). An OS home screen drawn on a canvas, NOT a
    73→ * dashboard of rounded boxes:
    74→ *
    75→ *   IDENTITY    PocketShell mark + wordmark + tagline · Info / Settings
    76→ *   FOUNDATION  the two environments — Terminal and Linux — as borderless
    77→ *               tone-step surfaces (§8)
    78→ *   TOOLS       "Your tools" — command apps as icon + label launcher entries
    79→ *               or the lightweight inline empty state (§6/§9)
    80→ *   SESSIONS    flat continuation rows between hairline dividers (§2a)
    81→ *   FLOATING    the single-purpose create control (§7)
    82→ *
    83→ * Surface rules (§3/§4): a surface is drawn ONLY for a real object — an
    84→ * environment, the CLI Apps menu, the floating control, an actionable banner,
    85→ * or a pressed row/tile. Text groupings are separated by spacing, section
    86→ * labels and hairline dividers, never containers. Packages are infrastructure
    87→ * and keep exactly ONE quiet affordance on this page in every state.
    88→ */
    89→@Composable
    90→fun HomeScreen(
    91→    terminalViewModel: TerminalViewModel,
    92→    activeSessions: List<TerminalSessionManager.SessionEntry>,
    93→    runtimeState: RuntimeState,
    94→    launchError: String?,
    95→    onDismissLaunchError: () -> Unit,
    96→    onOpenTerminal: () -> Unit,
    97→    onNewTerminal: () -> Unit,
    98→    onOpenLinuxShell: () -> Unit,
    99→    onOpenSession: (Long) -> Unit,
   100→    onOpenCommandApp: (CommandApp) -> Unit,
   101→    onExplorePackages: () -> Unit,
   102→    onOpenSettings: () -> Unit,
   103→    onOpenDiagnostics: () -> Unit,
   104→    modifier: Modifier = Modifier,
   105→) {
   106→    val commandApps by terminalViewModel.commandApps.collectAsStateWithLifecycle()
   107→    val verifyingApp by terminalViewModel.verifyingApp.collectAsStateWithLifecycle()
   108→    val operation by terminalViewModel.packageOperation.collectAsStateWithLifecycle()
   109→    val creating by terminalViewModel.creating.collectAsStateWithLifecycle()
   110→
   111→    // Probe availability whenever Home is visible with a READY runtime …
   112→    LaunchedEffect(runtimeState) {
   113→        if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
   114→    }
   115→    // … and after any package operation lands (a fresh install may add apps).
   116→    LaunchedEffect(operation?.id, operation?.state) {
   117→        val state = operation?.state
   118→        if (state == PackageOperationState.SUCCESS || state == PackageOperationState.FAILED) {
   119→            if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
   120→        }
   121→    }
   122→
   123→    // The floating control creates sessions — nothing else (§7). Command apps
   124→    // launch from the tools grid and the CLI Apps menu, never from here.
   125→    val quickActions = remember(runtimeState, creating, onNewTerminal, onOpenLinuxShell) {
   126→        buildList {
   127→            if (runtimeState == RuntimeState.READY) {
   128→                add(
   129→                    QuickAction(
   130→                        id = "new-linux",
   131→                        label = "New Linux session",
   132→                        enabled = !creating,
   133→                    ) { onOpenLinuxShell() },
   134→                )
   135→            }
   136→            add(
   137→                QuickAction(
   138→                    id = "new-terminal",
   139→                    label = "New Terminal",
   140→                    enabled = !creating,
   141→                ) { onNewTerminal() },
   142→            )
   143→        }
   144→    }
   145→
   146→    BoxWithConstraints(
   147→        modifier = modifier
   148→            .fillMaxSize()
   149→            .background(TerminalTheme.screenBg),
   150→    ) {
   151→        val columns = when {
   152→            maxWidth >= 840.dp -> 6
   153→            maxWidth >= 600.dp -> 4
   154→            else -> 3
   155→        }
   156→        Column(
   157→            modifier = Modifier
   158→                .fillMaxSize()
   159→                .verticalScroll(rememberScrollState())
   160→                .statusBarsPadding()
   161→                .navigationBarsPadding(),
   162→        ) {
   163→            Column(
   164→                modifier = Modifier
   165→                    .fillMaxWidth()
   166→                    .widthIn(max = HomeTokens.contentMaxWidth)
   167→                    .align(Alignment.CenterHorizontally),
   168→            ) {
   169→                Spacer(Modifier.height(12.dp))
   170→                BrandHeader(onOpenDiagnostics, onOpenSettings)
   171→                Spacer(Modifier.height(24.dp))
   172→
   173→                if (launchError != null) {
   174→                    LaunchErrorBanner(
   175→                        message = launchError,
   176→                        onDismiss = onDismissLaunchError,
   177→                        onOpenDiagnostics = { onDismissLaunchError(); onOpenDiagnostics() },
   178→                    )
   179→                    Spacer(Modifier.height(20.dp))
   180→                }
   181→
   182→                EnvironmentLaunchers(
   183→                    runtimeState = runtimeState,
   184→                    runningSessions = activeSessions.count { !it.isFinished },
   185→                    onOpenTerminal = onOpenTerminal,
   186→                    onOpenLinuxShell = onOpenLinuxShell,
   187→                    onOpenDiagnostics = onOpenDiagnostics,
   188→                )
   189→
   190→                // ONE CLI control in the header area (§5) — rendered only when
   191→                // the guest actually confirmed apps; never a dead button.
   192→                if (commandApps.apps.isNotEmpty()) {
   193→                    Spacer(Modifier.height(16.dp))
   194→                    CliAppsMenu(
   195→                        apps = commandApps.apps,
   196→                        onOpenCommandApp = onOpenCommandApp,
   197→                    )
   198→                    Spacer(Modifier.height(16.dp))
   199→                } else {
   200→                    Spacer(Modifier.height(24.dp))
   201→                }
   202→                SectionDivider()
   203→                Spacer(Modifier.height(16.dp))
   204→
   205→                ToolsSection(
   206→                    state = commandApps,
   207→                    verifyingApp = verifyingApp,
   208→                    runtimeReady = runtimeState == RuntimeState.READY,
   209→                    columns = columns,
   210→                    onOpenCommandApp = onOpenCommandApp,
   211→                    onExplorePackages = onExplorePackages,
   212→                )
   213→
   214→                if (activeSessions.isNotEmpty()) {
   215→                    Spacer(Modifier.height(24.dp))
   216→                    SectionDivider()
   217→                    Spacer(Modifier.height(12.dp))
   218→                    SessionsSection(activeSessions, onOpenSession)
   219→                }
   220→
   221→                if (commandApps.apps.isNotEmpty()) {
   222→                    // Exactly ONE packages affordance when tools exist — the
   223→                    // empty state carries it when they don't (§9).
   224→                    Spacer(Modifier.height(20.dp))
   225→                    PackagesFooterLink(onExplorePackages)
   226→                }
   227→
   228→                // clearance for the floating create control
   229→                Spacer(Modifier.height(140.dp))
   230→            }
   231→        }
   232→
   233→        QuickActionsOverlay(actions = quickActions)
   234→    }
   235→}

   418→@Composable
   419→private fun LinuxTile(runtimeState: RuntimeState, onClick: () -> Unit, modifier: Modifier = Modifier) {
   420→    // Honest gate (M2 architecture): READY enters the guest, everything else
   421→    // routes to Diagnostics where install/retry/repair actually live.
   422→    val stateLine = when (runtimeState) {
   423→        RuntimeState.READY -> "Alpine · ready"
   424→        RuntimeState.NOT_INSTALLED -> "Not installed yet"
   425→        RuntimeState.DOWNLOADING,
   426→        RuntimeState.VERIFYING,
   427→        RuntimeState.EXTRACTING,
   428→        RuntimeState.CONFIGURING,
   429→        -> "Installing…"
   430→        RuntimeState.FAILED -> "Install failed"
   431→        RuntimeState.REPAIR_REQUIRED -> "Repair needed"
   432→        RuntimeState.UNSUPPORTED_ABI -> "No arm64 CPU"
   433→    }
   434→    PressableScale(
   435→        onClick = onClick,
   436→        onClickLabel = if (runtimeState == RuntimeState.READY) "Open the Linux environment" else "Open Diagnostics",
   437→        modifier = modifier,
   438→    ) {
   439→        Column(
   440→            modifier = Modifier
   441→                .fillMaxWidth()
   442→                .height(160.dp)
   443→                .clip(RoundedCornerShape(HomeTokens.heroRadius))
   444→                .background(HomeTokens.surfaceEnv)
   445→                .padding(16.dp),

   447→            Row(modifier = Modifier.fillMaxWidth(), verticalAlignment = Alignment.Top) {
   448→                MountainMark(size = 32.dp)
   449→                Spacer(Modifier.weight(1f))
   450→                if (runtimeState == RuntimeState.READY) {
   451→                    ReadyDot()
   452→                }
   453→            }
   454→            Spacer(Modifier.weight(1f))
   455→            Text(
   456→                text = "Linux",
   457→                style = MaterialTheme.typography.titleLarge,
   458→                fontWeight = FontWeight.SemiBold,
   459→                color = HomeTokens.textPrimary,
   460→            )
   461→            Spacer(Modifier.height(2.dp))

   471→                    style = MaterialTheme.typography.bodySmall,
   472→                    color = HomeTokens.textDim.copy(alpha = 0.75f),
   473→                    maxLines = 1,
   474→                )
   475→            }
   476→        }
   477→    }

   593→    runtimeReady: Boolean,
   594→    columns: Int,
   595→    onOpenCommandApp: (CommandApp) -> Unit,
   596→    onExplorePackages: () -> Unit,
   597→) {
   598→    Column(modifier = Modifier.fillMaxWidth().padding(horizontal = 20.dp)) {
   599→        HomeSectionLabel("Your tools")
   600→        Spacer(Modifier.height(14.dp))
   601→        when {
   602→            state.apps.isEmpty() && !state.probeError.isNullOrEmpty() -> {
   603→                // Probe failed, nothing previously confirmed: the honest
   604→                // "could not check" — never a fake "none" (v0.4.4 rule).
   605→                Text(
   606→                    text = "App availability could not be checked — ${state.probeError}",
   607→                    style = MaterialTheme.typography.bodySmall,
   608→                    color = HomeTokens.textDim,
   609→                )
   610→            }
   611→            state.apps.isEmpty() && runtimeReady && !state.checked -> {
   612→                // Probe in flight: one quiet line, the launcher never flashes
   613→                // a fake "none installed" while the guest is being asked.
   614→                Row(verticalAlignment = Alignment.CenterVertically) {
   615→                    CircularProgressIndicator(
   616→                        modifier = Modifier.size(14.dp),
   617→                        strokeWidth = 2.dp,
   618→                        color = HomeTokens.accent,
   619→                    )
   620→                    Spacer(Modifier.width(10.dp))
   621→                    Text(
   622→                        text = "Checking the Linux environment…",
   623→                        style = MaterialTheme.typography.bodySmall,
   624→                        color = HomeTokens.textDim,
   625→                    )
   626→                }
   627→            }
   628→            state.apps.isEmpty() -> {
   629→                EmptyToolsState(runtimeReady = runtimeReady, onExplorePackages = onExplorePackages)
   630→            }
   631→            else -> {
   632→                state.apps.chunked(columns).forEach { rowApps ->
   633→                    Row(
   634→                        modifier = Modifier.fillMaxWidth().padding(vertical = 6.dp),
   635→                    ) {
   636→                        rowApps.forEach { app ->
   637→                            CommandAppEntry(
   638→                                app = app,
   639→                                verifying = verifyingApp == app.displayName,
   640→                                onClick = { onOpenCommandApp(app) },
   641→                                modifier = Modifier.weight(1f),
   642→                            )
   643→                        }
   644→                        repeat(columns - rowApps.size) {
   645→                            Spacer(Modifier.weight(1f))
   646→                        }
   647→                    }
   648→                }
   649→                if (state.probeError != null) {
   650→                    Spacer(Modifier.height(10.dp))
   651→                    Text(
   652→                        text = "Availability last checked before an error: ${state.probeError}",
   653→                        style = MaterialTheme.typography.bodySmall,
   654→                        color = HomeTokens.textDim.copy(alpha = 0.8f),
   655→                    )
   656→                }
   657→            }
   658→        }
   659→    }
   660→}
   661→
   662→/**
   663→ * One launcher entry: icon + label — an application on an OS home screen, not
   664→ * a card (§6). The monogram plate is a borderless tone step; press feedback
   665→ * is the soft scale, nothing draws a box.
   666→ */
   667→@Composable
   668→private fun CommandAppEntry(
   669→    app: CommandApp,
   670→    verifying: Boolean,
   671→    onClick: () -> Unit,
   672→    modifier: Modifier = Modifier,
   673→) {
   674→    PressableScale(
   675→        onClick = onClick,
   676→        onClickLabel = "Open ${app.displayName}",
   677→        modifier = modifier,
   678→        pressedScale = 0.96f,
   679→    ) {
   680→        Column(
   681→            modifier = Modifier
   682→                .fillMaxWidth()
   683→                .padding(vertical = 6.dp, horizontal = 4.dp),
   684→            horizontalAlignment = Alignment.CenterHorizontally,
   685→        ) {
   686→            if (verifying) {
   687→                Box(
   688→                    modifier = Modifier.size(52.dp),
   689→                    contentAlignment = Alignment.Center,
   690→                ) {
   691→                    CircularProgressIndicator(
   692→                        modifier = Modifier.size(20.dp),
   693→                        strokeWidth = 2.dp,
   694→                        color = HomeTokens.accent,
   695→                    )
   696→                }
   697→            } else {
   698→                MonogramTile(monogram = app.monogram, size = 52.dp)
   699→            }
   700→            Spacer(Modifier.height(8.dp))
   701→            Text(
   702→                text = app.displayName,
   703→                style = MaterialTheme.typography.labelMedium,
   704→                color = HomeTokens.textPrimary,
   705→                maxLines = 1,
   706→                overflow = TextOverflow.Ellipsis,
   707→                textAlign = TextAlign.Center,
   708→            )
   709→        }
   710→    }
   711→}

   713→/**
   714→ * The empty launcher (§9): three quiet text lines directly on the canvas —
   715→ * no container, no placeholder icons, nothing that dominates. This carries
   716→ * the page's ONLY "Explore packages" affordance in this state.

   718→@Composable
   719→private fun EmptyToolsState(runtimeReady: Boolean, onExplorePackages: () -> Unit) {
   720→    Column(modifier = Modifier.fillMaxWidth()) {
   721→        Text(
   722→            text = "No CLI apps yet.",
   723→            style = MaterialTheme.typography.bodyMedium,
   724→            color = HomeTokens.textPrimary,
   725→        )
   726→        Spacer(Modifier.height(4.dp))
   727→        Text(
   728→            text = if (runtimeReady) {
   729→                "Install an interactive command application and it will appear here."
   730→            } else {
   731→                "Install the Linux runtime first — interactive command apps live " +
   732→                    "inside your Linux environment and launch from here."
   733→            },
   734→            style = MaterialTheme.typography.bodySmall,
   735→            color = HomeTokens.textDim,
   736→        )
   737→        Text(
   738→            text = "Explore packages",
   739→            style = MaterialTheme.typography.labelLarge,
   740→            color = HomeTokens.accent,
   741→            modifier = Modifier
   742→                .padding(top = 6.dp, start = 4.dp, end = 4.dp, bottom = 4.dp)
   743→                .clickable(role = Role.Button) { onExplorePackages() }
   744→                .padding(horizontal = 8.dp, vertical = 8.dp),
   745→        )
   746→    }

   751→/**
   752→ * Flat continuation rows (§2a) — running sessions stay visible without
   753→ * dominating: dot + label + mono id between hairline dividers. No boxes at
   754→ * rest; a pressed row is the only surface this section ever draws. Capped;
   755→ * the rest remain in the Terminal's tabs.
   756→ */
   757→@Composable
   758→private fun SessionsSection(
   759→    sessions: List<TerminalSessionManager.SessionEntry>,
   760→    onOpenSession: (Long) -> Unit,
   761→) {
   762→    Column(modifier = Modifier.fillMaxWidth().padding(horizontal = 20.dp)) {
   763→        HomeSectionLabel("Sessions")
   764→        Spacer(Modifier.height(6.dp))
   765→        val visible = sessions.take(4)
   766→        visible.forEachIndexed { index, entry ->
   767→            val finished = entry.isFinished
   768→            val interaction = remember { MutableInteractionSource() }
   769→            val pressed by interaction.collectIsPressedAsState()
   770→            Row(
   771→                modifier = Modifier
   772→                    .fillMaxWidth()
   773→                    .clip(RoundedCornerShape(10.dp))
   774→                    .background(if (pressed) HomeTokens.surfaceBanner else Color.Transparent)
   775→                    .clickable(
   776→                        interactionSource = interaction,
   777→                        indication = null,
   778→                        role = Role.Button,
   779→                        onClickLabel = "Return to session",
   780→                    ) { onOpenSession(entry.id) }
   781→                    .padding(horizontal = 10.dp, vertical = 12.dp),
   782→                verticalAlignment = Alignment.CenterVertically,
   783→            ) {
   784→                Box(
   785→                    modifier = Modifier
   786→                        .size(8.dp)
   787→                        .background(
   788→                            if (finished) HomeTokens.hairline else HomeTokens.runningGreen,
   789→                            CircleShape,
   790→                        ),
   791→                )
   792→                Spacer(Modifier.width(12.dp))
   793→                Text(
   794→                    text = entry.displayLabel + if (finished) " (exited)" else "",
   795→                    style = MaterialTheme.typography.bodyMedium,
   796→                    color = if (finished) HomeTokens.textDim else HomeTokens.textPrimary,
   797→                    maxLines = 1,
   798→                    overflow = TextOverflow.Ellipsis,
   799→                    modifier = Modifier.weight(1f),
   800→                )
   801→                Text(
   802→                    text = "#${entry.id}",
   803→                    fontFamily = TerminalTheme.mono,
   804→                    fontSize = 11.sp,
   805→                    color = HomeTokens.textDim.copy(alpha = 0.7f),
   806→                )
   807→            }
   808→            if (index != visible.lastIndex) {
   809→                HorizontalDivider(
   810→                    modifier = Modifier.padding(start = 30.dp),
   811→                    color = HomeTokens.hairline.copy(alpha = 0.6f),
   812→                )
   813→            }
   814→        }
   815→        if (sessions.size > 4) {
   816→            Spacer(Modifier.height(6.dp))
   817→            Text(
   818→                text = "+${sessions.size - 4} more in Terminal",
   819→                style = MaterialTheme.typography.bodySmall,
   820→                color = HomeTokens.textDim.copy(alpha = 0.8f),
   821→                modifier = Modifier.padding(start = 10.dp),
   822→            )
   823→        }
   824→    }
   825→}

   853→// ------------------------------------------------------------------- helpers
   854→
   855→/** Soft press feedback (80ms scale) shared by the launch surfaces. */
   856→@Composable
   857→private fun PressableScale(
   858→    onClick: () -> Unit,
   859→    onClickLabel: String?,
   860→    modifier: Modifier = Modifier,
   861→    pressedScale: Float = 0.98f,
   862→    content: @Composable () -> Unit,
   863→) {
   864→    val interaction = remember { MutableInteractionSource() }
   865→    val pressed by interaction.collectIsPressedAsState()
   866→    val scale by animateFloatAsState(
   867→        targetValue = if (pressed) pressedScale else 1f,
   868→        animationSpec = tween(80, easing = FastOutSlowInEasing),
   869→        label = "homePressScale",
   870→    )
   871→    Box(
   872→        modifier = modifier
   873→            .graphicsLayer { scaleX = scale; scaleY = scale }
   874→            .clickable(
   875→                interactionSource = interaction,
   876→                indication = null,
   877→                role = Role.Button,
   878→                onClickLabel = onClickLabel,
   879→            ) { onClick() },
   880→    ) {
   881→        content()
   882→    }
   883→}
