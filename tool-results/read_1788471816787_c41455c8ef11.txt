    59→/**
    60→ * PocketShell Home — the launcher of the PocketShell environment
    61→ * (docs/PHASE-3.2-DESIGN.md). An OS home screen, not a dashboard:
    62→ *
    63→ *   TOP      PocketShell identity · minimal system actions
    64→ *   CENTER   the two foundations — Terminal and Linux
    65→ *   BELOW    command-launchable apps (launcher grid) or the honest empty state
    66→ *   QUIET    compact running-session continuation area
    67→ *   FLOATING custom quick-action control (no bottom navigation bar)
    68→ *
    69→ * Packages are infrastructure and stay on the Packages screen; only apps the
    70→ * guest's login shell confirms appear here. Fixed Midnight Sapphire identity
    71→ * in every app theme — Home belongs beside the Phase 3.1 terminal.
    72→ */
    73→@Composable
    74→fun HomeScreen(
    75→    terminalViewModel: TerminalViewModel,
    76→    activeSessions: List<TerminalSessionManager.SessionEntry>,
    77→    runtimeState: RuntimeState,
    78→    launchError: String?,
    79→    onDismissLaunchError: () -> Unit,
    80→    onOpenTerminal: () -> Unit,
    81→    onNewTerminal: () -> Unit,
    82→    onOpenLinuxShell: () -> Unit,
    83→    onOpenSession: (Long) -> Unit,
    84→    onOpenCommandApp: (CommandApp) -> Unit,
    85→    onExplorePackages: () -> Unit,
    86→    onOpenSettings: () -> Unit,
    87→    onOpenDiagnostics: () -> Unit,
    88→    modifier: Modifier = Modifier,
    89→) {
    90→    val commandApps by terminalViewModel.commandApps.collectAsStateWithLifecycle()
    91→    val verifyingApp by terminalViewModel.verifyingApp.collectAsStateWithLifecycle()
    92→    val operation by terminalViewModel.packageOperation.collectAsStateWithLifecycle()
    93→    val creating by terminalViewModel.creating.collectAsStateWithLifecycle()
    94→
    95→    // Probe availability whenever Home is visible with a READY runtime …
    96→    LaunchedEffect(runtimeState) {
    97→        if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
    98→    }
    99→    // … and after any package operation lands (a fresh install may add apps).
   100→    LaunchedEffect(operation?.id, operation?.state) {
   101→        val state = operation?.state
   102→        if (state == PackageOperationState.SUCCESS || state == PackageOperationState.FAILED) {
   103→            if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
   104→        }
   105→    }
   106→
   107→    val quickActions = remember(
   108→        commandApps.apps, runtimeState, creating, onNewTerminal, onOpenLinuxShell, onOpenCommandApp,
   109→    ) {
   110→        buildList {
   111→            commandApps.apps.take(4).forEach { app ->
   112→                add(
   113→                    QuickAction(
   114→                        id = "app-${app.id}",
   115→                        label = app.displayName,
   116→                        glyph = QuickActionGlyph.APP,
   117→                        monogram = app.monogram,
   118→                        enabled = !creating,
   119→                    ) { onOpenCommandApp(app) },
   120→                )
   121→            }
   122→            if (runtimeState == RuntimeState.READY) {
   123→                add(
   124→                    QuickAction(
   125→                        id = "new-linux",
   126→                        label = "New Linux session",
   127→                        glyph = QuickActionGlyph.LINUX,
   128→                        enabled = !creating,
   129→                    ) { onOpenLinuxShell() },
   130→                )
   131→            }
   132→            add(
   133→                QuickAction(
   134→                    id = "new-terminal",
   135→                    label = "New Terminal",
   136→                    glyph = QuickActionGlyph.TERMINAL,
   137→                    enabled = !creating,
   138→                ) { onNewTerminal() },
   139→            )
   140→        }
   141→    }
   142→
   143→    BoxWithConstraints(
   144→        modifier = modifier
   145→            .fillMaxSize()
   146→            .background(TerminalTheme.screenBg),
   147→    ) {
   148→        val columns = when {
   149→            maxWidth >= 840.dp -> 6
   150→            maxWidth >= 600.dp -> 4
   151→            else -> 3
   152→        }
   153→        Column(
   154→            modifier = Modifier
   155→                .fillMaxSize()
   156→                .verticalScroll(rememberScrollState())
   157→                .statusBarsPadding()
   158→                .navigationBarsPadding(),
   159→        ) {
   160→            Column(
   161→                modifier = Modifier
   162→                    .fillMaxWidth()
   163→                    .widthIn(max = HomeTokens.contentMaxWidth)
   164→                    .align(Alignment.CenterHorizontally),
   165→            ) {
   166→                Spacer(Modifier.height(12.dp))
   167→                BrandHeader(onOpenDiagnostics, onOpenSettings)
   168→                Spacer(Modifier.height(28.dp))
   169→
   170→                if (launchError != null) {
   171→                    LaunchErrorBanner(
   172→                        message = launchError,
   173→                        onDismiss = onDismissLaunchError,
   174→                        onOpenDiagnostics = { onDismissLaunchError(); onOpenDiagnostics() },
   175→                    )
   176→                    Spacer(Modifier.height(20.dp))
   177→                }
   178→
   179→                EnvironmentLaunchers(
   180→                    runtimeState = runtimeState,
   181→                    runningSessions = activeSessions.count { !it.isFinished },
   182→                    onOpenTerminal = onOpenTerminal,
   183→                    onOpenLinuxShell = onOpenLinuxShell,
   184→                    onOpenDiagnostics = onOpenDiagnostics,
   185→                )
   186→                Spacer(Modifier.height(32.dp))
   187→
   188→                CommandAppsArea(
   189→                    state = commandApps,
   190→                    verifyingApp = verifyingApp,
   191→                    runtimeReady = runtimeState == RuntimeState.READY,
   192→                    columns = columns,
   193→                    onOpenCommandApp = onOpenCommandApp,
   194→                    onExplorePackages = onExplorePackages,
   195→                )
   196→
   197→                if (activeSessions.isNotEmpty()) {
   198→                    Spacer(Modifier.height(28.dp))
   199→                    SessionsContinuationArea(activeSessions, onOpenSession)
   200→                }
   201→
   202→                Spacer(Modifier.height(16.dp))
   203→                PackagesFooterLink(onExplorePackages)
   204→                // clearance for the floating quick-action control
   205→                Spacer(Modifier.height(140.dp))
   206→            }
   207→        }
   208→
   209→        QuickActionsOverlay(actions = quickActions)
   210→    }
   211→}

   377→@Composable
   378→private fun LinuxTile(runtimeState: RuntimeState, onClick: () -> Unit, modifier: Modifier = Modifier) {
   379→    // Honest gate (M2 architecture): READY enters the guest, everything else
   380→    // routes to Diagnostics where install/retry/repair actually live.
   381→    val stateLine = when (runtimeState) {
   382→        RuntimeState.READY -> "Alpine Linux · ready"
   383→        RuntimeState.NOT_INSTALLED -> "Not installed yet"
   384→        RuntimeState.DOWNLOADING,
   385→        RuntimeState.VERIFYING,
   386→        RuntimeState.EXTRACTING,
   387→        RuntimeState.CONFIGURING,
   388→        -> "Install in progress"
   389→        RuntimeState.FAILED -> "Install failed"
   390→        RuntimeState.REPAIR_REQUIRED -> "Repair needed"
   391→        RuntimeState.UNSUPPORTED_ABI -> "No arm64 CPU"
   392→    }
   393→    val supportLine = when (runtimeState) {
   394→        RuntimeState.READY -> "Enter the guest shell"
   395→        RuntimeState.UNSUPPORTED_ABI -> "Runtime unavailable on this device"
   396→        else -> "Set up or repair from Diagnostics"
   397→    }
   398→    PressableScale(
   399→        onClick = onClick,
   400→        onClickLabel = if (runtimeState == RuntimeState.READY) "Open the Linux environment" else "Open Diagnostics",
   401→        modifier = modifier,
   402→    ) {
   403→        Column(
   404→            modifier = Modifier
   405→                .fillMaxWidth()
   406→                .height(168.dp)
   407→                .clip(RoundedCornerShape(HomeTokens.heroRadius))
   408→                .background(HomeTokens.surfaceEnv)
   409→                .border(1.dp, HomeTokens.hairline, RoundedCornerShape(HomeTokens.heroRadius))
   410→                .padding(18.dp),
   411→        ) {
   412→            Row(modifier = Modifier.fillMaxWidth(), verticalAlignment = Alignment.Top) {
   413→                MountainMark(size = 36.dp)
   414→                Spacer(Modifier.weight(1f))
   415→                if (runtimeState == RuntimeState.READY) {
   416→                    ReadyDot()
   417→                }
   418→            }
   419→            Spacer(Modifier.weight(1f))
   420→            Text(
   421→                text = "Linux",
   422→                style = MaterialTheme.typography.titleLarge,
   423→                fontWeight = FontWeight.SemiBold,
   424→                color = HomeTokens.textPrimary,
   425→            )
   426→            Spacer(Modifier.height(2.dp))
   427→            Text(
   428→                text = stateLine,
   429→                style = MaterialTheme.typography.bodySmall,
   430→                color = if (runtimeState == RuntimeState.READY) HomeTokens.accent else HomeTokens.textDim,

   442→    }

   476→@Composable
   477→private fun CommandAppsArea(
   478→    state: TerminalViewModel.CommandAppsState,
   479→    verifyingApp: String?,
   480→    runtimeReady: Boolean,
   481→    columns: Int,
   482→    onOpenCommandApp: (CommandApp) -> Unit,
   483→    onExplorePackages: () -> Unit,
   484→) {
   485→    Column(modifier = Modifier.fillMaxWidth().padding(horizontal = 20.dp)) {
   486→        if (state.apps.isEmpty() && !state.probeError.isNullOrEmpty()) {
   487→            // Probe failed: the honest "could not check" — never a fake "none".
   488→            Column(
   489→                modifier = Modifier
   490→                    .fillMaxWidth()
   491→                    .clip(RoundedCornerShape(HomeTokens.chipRadius))
   492→                    .background(HomeTokens.surfaceBanner)
   493→                    .border(1.dp, HomeTokens.hairline, RoundedCornerShape(HomeTokens.chipRadius))
   494→                    .padding(18.dp),
   495→            ) {
   496→                HomeSectionLabel("Command apps")
   497→                Spacer(Modifier.height(10.dp))
   498→                Text(
   499→                    text = "App availability could not be checked right now — ${state.probeError}",
   500→                    style = MaterialTheme.typography.bodySmall,
   501→                    color = HomeTokens.textDim,
   502→                )
   503→            }
   504→        } else if (state.apps.isEmpty() && runtimeReady && !state.checked) {
   505→            // Probe in flight: a quiet honest placeholder — the launcher never
   506→            // flashes a fake "none installed" while the guest is being asked.
   507→            Column(
   508→                modifier = Modifier
   509→                    .fillMaxWidth()
   510→                    .clip(RoundedCornerShape(HomeTokens.chipRadius))
   511→                    .background(HomeTokens.surfaceBanner)
   512→                    .border(1.dp, HomeTokens.hairline, RoundedCornerShape(HomeTokens.chipRadius))
   513→                    .padding(18.dp),
   514→            ) {
   515→                HomeSectionLabel("Command apps")
   516→                Spacer(Modifier.height(12.dp))
   517→                Row(verticalAlignment = Alignment.CenterVertically) {
   518→                    CircularProgressIndicator(
   519→                        modifier = Modifier.size(14.dp),
   520→                        strokeWidth = 2.dp,
   521→                        color = HomeTokens.accent,
   522→                    )
   523→                    Spacer(Modifier.width(10.dp))
   524→                    Text(
   525→                        text = "Checking the Linux environment…",
   526→                        style = MaterialTheme.typography.bodySmall,
   527→                        color = HomeTokens.textDim,
   528→                    )
   529→                }
   530→            }
   531→        } else if (state.apps.isEmpty()) {
   532→            // Real empty answer (or runtime not ready yet): the launcher's
   533→            // beautiful empty state — the body text states which truth applies.
   534→            EmptyAppsState(runtimeReady = runtimeReady, onExplorePackages = onExplorePackages)
   535→        } else {
   536→            HomeSectionLabel("Command apps")
   537→            Spacer(Modifier.height(14.dp))
   538→            state.apps.chunked(columns).forEach { rowApps ->
   539→                Row(
   540→                    modifier = Modifier.fillMaxWidth().padding(vertical = 6.dp),
   541→                ) {
   542→                    rowApps.forEach { app ->
   543→                        CommandAppTile(
   544→                            app = app,
   545→                            verifying = verifyingApp == app.displayName,
   546→                            onClick = { onOpenCommandApp(app) },
   547→                            modifier = Modifier.weight(1f),
   548→                        )
   549→                    }
   550→                    repeat(columns - rowApps.size) {
   551→                        Spacer(Modifier.weight(1f))
   552→                    }
   553→                }
   554→            }
   555→            if (state.probeError != null) {
   556→                Spacer(Modifier.height(10.dp))
   557→                Text(
   558→                    text = "Availability last checked before an error: ${state.probeError}",
   559→                    style = MaterialTheme.typography.bodySmall,
   560→                    color = HomeTokens.textDim.copy(alpha = 0.8f),
   561→                )
   562→            }
   563→        }
   564→    }
   565→}

   750→/** Soft press feedback (80ms scale) shared by the launch surfaces. */
   751→@Composable
   752→private fun PressableScale(
   753→    onClick: () -> Unit,
   754→    onClickLabel: String?,
   755→    modifier: Modifier = Modifier,
   756→    pressedScale: Float = 0.98f,
   757→    content: @Composable () -> Unit,
   758→) {
   759→    val interaction = remember { MutableInteractionSource() }
   760→    val pressed by interaction.collectIsPressedAsState()
   761→    val scale by animateFloatAsState(
   762→        targetValue = if (pressed) pressedScale else 1f,
   763→        animationSpec = tween(80, easing = FastOutSlowInEasing),
   764→        label = "homePressScale",
   765→    )
   766→    Box(
   767→        modifier = modifier
   768→            .graphicsLayer { scaleX = scale; scaleY = scale }
   769→            .clickable(
   770→                interactionSource = interaction,
   771→                indication = null,
   772→                role = Role.Button,
   773→                onClickLabel = onClickLabel,
   774→            ) { onClick() },
   775→    ) {
   776→        content()
   777→    }
   778→}
