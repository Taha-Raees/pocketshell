    59→    59→/**
    60→    60→ * PocketShell Home — the launcher of the PocketShell environment
    61→    61→ * (docs/PHASE-3.2-DESIGN.md). An OS home screen, not a dashboard:
    62→    62→ *
    63→    63→ *   TOP      PocketShell identity · minimal system actions
    64→    64→ *   CENTER   the two foundations — Terminal and Linux
    65→    65→ *   BELOW    command-launchable apps (launcher grid) or the honest empty state
    66→    66→ *   QUIET    compact running-session continuation area
    67→    67→ *   FLOATING custom quick-action control (no bottom navigation bar)
    68→    68→ *
    69→    69→ * Packages are infrastructure and stay on the Packages screen; only apps the
    70→    70→ * guest's login shell confirms appear here. Fixed Midnight Sapphire identity
    71→    71→ * in every app theme — Home belongs beside the Phase 3.1 terminal.
    72→    72→ */
    73→    73→@Composable
    74→    74→fun HomeScreen(
    75→    75→    terminalViewModel: TerminalViewModel,
    76→    76→    activeSessions: List<TerminalSessionManager.SessionEntry>,
    77→    77→    runtimeState: RuntimeState,
    78→    78→    launchError: String?,
    79→    79→    onDismissLaunchError: () -> Unit,
    80→    80→    onOpenTerminal: () -> Unit,
    81→    81→    onNewTerminal: () -> Unit,
    82→    82→    onOpenLinuxShell: () -> Unit,
    83→    83→    onOpenSession: (Long) -> Unit,
    84→    84→    onOpenCommandApp: (CommandApp) -> Unit,
    85→    85→    onExplorePackages: () -> Unit,
    86→    86→    onOpenSettings: () -> Unit,
    87→    87→    onOpenDiagnostics: () -> Unit,
    88→    88→    modifier: Modifier = Modifier,
    89→    89→) {
    90→    90→    val commandApps by terminalViewModel.commandApps.collectAsStateWithLifecycle()
    91→    91→    val verifyingApp by terminalViewModel.verifyingApp.collectAsStateWithLifecycle()
    92→    92→    val operation by terminalViewModel.packageOperation.collectAsStateWithLifecycle()
    93→    93→    val creating by terminalViewModel.creating.collectAsStateWithLifecycle()
    94→    94→
    95→    95→    // Probe availability whenever Home is visible with a READY runtime …
    96→    96→    LaunchedEffect(runtimeState) {
    97→    97→        if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
    98→    98→    }
    99→    99→    // … and after any package operation lands (a fresh install may add apps).
   100→   100→    LaunchedEffect(operation?.id, operation?.state) {
   101→   101→        val state = operation?.state
   102→   102→        if (state == PackageOperationState.SUCCESS || state == PackageOperationState.FAILED) {
   103→   103→            if (runtimeState == RuntimeState.READY) terminalViewModel.refreshCommandApps()
   104→   104→        }
   105→   105→    }
   106→   106→
   107→   107→    val quickActions = remember(
   108→   108→        commandApps.apps, runtimeState, creating, onNewTerminal, onOpenLinuxShell, onOpenCommandApp,
   109→   109→    ) {
   110→   110→        buildList {
   111→   111→            commandApps.apps.take(4).forEach { app ->
   112→   112→                add(
   113→   113→                    QuickAction(
   114→   114→                        id = "app-${app.id}",
   115→   115→                        label = app.displayName,
   116→   116→                        glyph = QuickActionGlyph.APP,
   117→   117→                        monogram = app.monogram,
   118→   118→                        enabled = !creating,
   119→   119→                    ) { onOpenCommandApp(app) },
   120→   120→                )
   121→   121→            }
   122→   122→            if (runtimeState == RuntimeState.READY) {
   123→   123→                add(
   124→   124→                    QuickAction(
   125→   125→                        id = "new-linux",
   126→   126→                        label = "New Linux session",
   127→   127→                        glyph = QuickActionGlyph.LINUX,
   128→   128→                        enabled = !creating,
   129→   129→                    ) { onOpenLinuxShell() },
   130→   130→                )
   131→   131→            }
   132→   132→            add(
   133→   133→                QuickAction(
   134→   134→                    id = "new-terminal",
   135→   135→                    label = "New Terminal",
   136→   136→                    glyph = QuickActionGlyph.TERMINAL,
   137→   137→                    enabled = !creating,
   138→   138→                ) { onNewTerminal() },
   139→   139→            )
   140→   140→        }
   141→   141→    }
   142→   142→
   143→   143→    BoxWithConstraints(
   144→   144→        modifier = modifier
   145→   145→            .fillMaxSize()
   146→   146→            .background(TerminalTheme.screenBg),
   147→   147→    ) {
   148→   148→        val columns = when {
   149→   149→            maxWidth >= 840.dp -> 6
   150→   150→            maxWidth >= 600.dp -> 4
   151→   151→            else -> 3
   152→   152→        }
   153→   153→        Column(
   154→   154→            modifier = Modifier
   155→   155→                .fillMaxSize()
   156→   156→                .verticalScroll(rememberScrollState())
   157→   157→                .statusBarsPadding()
   158→   158→                .navigationBarsPadding(),
   159→   159→        ) {
   160→   160→            Column(
   161→   161→                modifier = Modifier
   162→   162→                    .fillMaxWidth()
   163→   163→                    .widthIn(max = HomeTokens.contentMaxWidth)
   164→   164→                    .align(Alignment.CenterHorizontally),
   165→   165→            ) {
   166→   166→                Spacer(Modifier.height(12.dp))
   167→   167→                BrandHeader(onOpenDiagnostics, onOpenSettings)
   168→   168→                Spacer(Modifier.height(28.dp))
   169→   169→
   170→   170→                if (launchError != null) {
   171→   171→                    LaunchErrorBanner(
   172→   172→                        message = launchError,
   173→   173→                        onDismiss = onDismissLaunchError,
   174→   174→                        onOpenDiagnostics = { onDismissLaunchError(); onOpenDiagnostics() },
   175→   175→                    )
   176→   176→                    Spacer(Modifier.height(20.dp))
   177→   177→                }
   178→   178→
   179→   179→                EnvironmentLaunchers(
   180→   180→                    runtimeState = runtimeState,
   181→   181→                    runningSessions = activeSessions.count { !it.isFinished },
   182→   182→                    onOpenTerminal = onOpenTerminal,
   183→   183→                    onOpenLinuxShell = onOpenLinuxShell,
   184→   184→                    onOpenDiagnostics = onOpenDiagnostics,
   185→   185→                )
   186→   186→                Spacer(Modifier.height(32.dp))
   187→   187→
   188→   188→                CommandAppsArea(
   189→   189→                    state = commandApps,
   190→   190→                    verifyingApp = verifyingApp,
   191→   191→                    runtimeReady = runtimeState == RuntimeState.READY,
   192→   192→                    columns = columns,
   193→   193→                    onOpenCommandApp = onOpenCommandApp,
   194→   194→                    onExplorePackages = onExplorePackages,
   195→   195→                )
   196→   196→
   197→   197→                if (activeSessions.isNotEmpty()) {
   198→   198→                    Spacer(Modifier.height(28.dp))
   199→   199→                    SessionsContinuationArea(activeSessions, onOpenSession)
   200→   200→                }
   201→   201→
   202→   202→                Spacer(Modifier.height(16.dp))
   203→   203→                PackagesFooterLink(onExplorePackages)
   204→   204→                // clearance for the floating quick-action control
   205→   205→                Spacer(Modifier.height(140.dp))
   206→   206→            }
   207→   207→        }
   208→   208→
   209→   209→        QuickActionsOverlay(actions = quickActions)
   210→   210→    }
   211→   211→}

   477→   477→private fun CommandAppsArea(

   750→   750→/** Soft press feedback (80ms scale) shared by the launch surfaces. */
   751→   751→@Composable
   752→   752→private fun PressableScale(
   753→   753→    onClick: () -> Unit,
   754→   754→    onClickLabel: String?,
   755→   755→    modifier: Modifier = Modifier,
   756→   756→    pressedScale: Float = 0.98f,
   757→   757→    content: @Composable () -> Unit,
   758→   758→) {
   759→   759→    val interaction = remember { MutableInteractionSource() }
   760→   760→    val pressed by interaction.collectIsPressedAsState()
   761→   761→    val scale by animateFloatAsState(
   762→   762→        targetValue = if (pressed) pressedScale else 1f,
   763→   763→        animationSpec = tween(80, easing = FastOutSlowInEasing),
   764→   764→        label = "homePressScale",
   765→   765→    )
   766→   766→    Box(
   767→   767→        modifier = modifier
   768→   768→            .graphicsLayer { scaleX = scale; scaleY = scale }
   769→   769→            .clickable(
   770→   770→                interactionSource = interaction,
   771→   771→                indication = null,
   772→   772→                role = Role.Button,
   773→   773→                onClickLabel = onClickLabel,
   774→   774→            ) { onClick() },
   775→   775→    ) {
   776→   776→        content()
   777→   777→    }
